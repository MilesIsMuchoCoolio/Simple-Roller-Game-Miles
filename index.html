var Draw = { canvas: null, ctx: null, cameraX: 0 }; Draw.setup = function () { Draw.canvas = document.getElementById("game"); Draw.ctx = Draw.canvas.getContext("2d"); }; Draw.updateCamera = function () { Draw.cameraX = Player.x - CONFIG.CANVAS_W / 2; if (Draw.cameraX < 0) { Draw.cameraX = 0; } var furthest = Math.max(0, Level.pixelWidth() - CONFIG.CANVAS_W); if (Draw.cameraX > furthest) { Draw.cameraX = furthest; } }; Draw.everything = function () { var ctx = Draw.ctx; ctx.fillStyle = "#fff"; ctx.fillRect(0, 0, CONFIG.CANVAS_W, CONFIG.CANVAS_H); ctx.save(); ctx.translate(-Draw.cameraX, 0); Draw.world(); Crumble.draw(); Combat.draw(); Draw.player(); ctx.restore(); }; Draw.world = function () { var size = CONFIG.TILE; var firstCol = Math.floor(Draw.cameraX / size) - 1; var lastCol = firstCol + Math.ceil(CONFIG.CANVAS_W / size) + 2; for (var row = 0; row < CONFIG.ROWS; row++) { for (var col = firstCol; col <= lastCol; col++) { var here = Level.charAt(col, row), x = col * size, y = row * size; if (here === "#") { Draw.block(x, y, size); } if (here === "^") { Draw.spike(x, y, size); } if (here === "C") { Draw.coin(x, y, size); } if (here === "A") { Draw.ammo(x, y, size); } if (here === "F") { Draw.finish(x, y, size); } } } }; Draw.block = function (x, y, size) { var c = Draw.ctx; c.fillStyle="#fff"; c.fillRect(x,y,size,size); c.strokeStyle="#000"; c.lineWidth=CONFIG.LINE_WIDTH; c.strokeRect(x+1.5,y+1.5,size-3,size-3); }; Draw.coin = function (x, y, size) { var c=Draw.ctx; c.fillStyle="#ffd400"; c.beginPath(); c.arc(x+size/2,y+size/2,size*.22,0,Math.PI*2); c.fill(); c.strokeStyle="#000"; c.lineWidth=3; c.stroke(); }; Draw.ammo = function (x, y, size) { var c=Draw.ctx; c.fillStyle="#58a6ff"; c.fillRect(x+10,y+8,20,24); c.strokeStyle="#000"; c.lineWidth=3; c.strokeRect(x+10,y+8,20,24); c.fillStyle="#000"; c.fillRect(x+15,y+3,10,8); }; Draw.spike = function (x,y,size) { var c=Draw.ctx; c.fillStyle="#000"; c.beginPath(); c.moveTo(x,y+size); c.lineTo(x+size/2,y); c.lineTo(x+size,y+size); c.closePath(); c.fill(); }; Draw.finish = function (x,y,size) { var c=Draw.ctx; c.fillStyle="#000"; c.fillRect(x+size/2-2,y,4,size); c.beginPath(); c.moveTo(x+size/2+2,y+4); c.lineTo(x+size-4,y+12); c.lineTo(x+size/2+2,y+20); c.fill(); }; Draw.player = function () { var c=Draw.ctx, x=Player.x+16, y=Player.y+8; c.strokeStyle="#111"; c.fillStyle="#4da3ff"; c.lineWidth=3; c.beginPath(); c.arc(x,y,8,0,Math.PI*2); c.fill(); c.stroke(); c.beginPath(); c.moveTo(x,y+8); c.lineTo(x,y+23); c.moveTo(x,y+12); c.lineTo(x-11,y+18); c.moveTo(x,y+12); c.lineTo(x+13,y+17); c.moveTo(x,y+23); c.lineTo(x-9,y+31); c.moveTo(x,y+23); c.lineTo(x+9,y+31); c.stroke(); c.strokeStyle="#000"; c.lineWidth=5; c.beginPath(); c.moveTo(x+9,y+14); c.lineTo(x+24,y+14); c.stroke(); };
